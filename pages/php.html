PHP Nima?

⏱ 8 daqiqa 🟡 O'rta 👁 6,180 marta ko'rilgan

PHP (PHP: Hypertext Preprocessor) — server tomonida ishlovchi skript tili. WordPress, Laravel va boshqa ko'plab mashhur platformalar PHP da yozilgan. Internetdagi saytlarning 77% dan ortiq qismi PHP dan foydalanadi.

💡 Eslatma: PHP kodi <?php ... ?> teglar ichida yoziladi va server tomonida bajariladi.

OOP — Klasslar

PHP (OOP) 
<?php
class Foydalanuvchi {
    // Xususiyatlar
    private string $ism;
    private string $email;
    protected int $yosh;

    // Konstruktor
    public function __construct(string $ism, string $email, int $yosh) {
        $this->ism = $ism;
        $this->email = $email;
        $this->yosh = $yosh;
    }

    // Metodlar
    public function getMalumat(): string {
        return "{$this->ism} ({$this->email}), yosh: {$this->yosh}";
    }

    public function voyagaYetganmi(): bool {
        return $this->yosh >= 18;
    }
}

// Meros
class Admin extends Foydalanuvchi {
    private array $ruxsatlar;

    public function __construct(string $ism, string $email, int $yosh, array $ruxsatlar) {
        parent::__construct($ism, $email, $yosh);
        $this->ruxsatlar = $ruxsatlar;
    }

    public function ruxsatBormi(string $ruxsat): bool {
        return in_array($ruxsat, $this->ruxsatlar);
    }
}

$user = new Foydalanuvchi("Ali", "ali@test.com", 25);
echo $user->getMalumat();

$admin = new Admin("Sarvar", "sarvar@test.com", 30, ["o'chirish", "tahrirlash"]);
echo $admin->ruxsatBormi("o'chirish") ? "Ruxsat bor" : "Ruxsat yo'q";
?>
