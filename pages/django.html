Python (models.py) 
# kurslar/models.py
from django.db import models
from django.contrib.auth.models import User

class Kategoriya(models.Model):
    nomi = models.CharField(max_length=100)
    slug = models.SlugField(unique=True)

    def __str__(self):
        return self.nomi

    class Meta:
        verbose_name_plural = "Kategoriyalar"

class Kurs(models.Model):
    DARAJA_CHOICES = [
        ('boshlangich', 'Boshlang\'ich'),
        ('orta', 'O\'rta'),
        ('ilgor', 'Ilg\'or'),
    ]

    sarlavha = models.CharField(max_length=200)
    tavsif = models.TextField()
    kategoriya = models.ForeignKey(Kategoriya, on_delete=models.CASCADE)
    muallif = models.ForeignKey(User, on_delete=models.SET_NULL, null=True)
    daraja = models.CharField(max_length=20, choices=DARAJA_CHOICES)
    yaratildi = models.DateTimeField(auto_now_add=True)
    yangilandi = models.DateTimeField(auto_now=True)
    nashr_etildi = models.BooleanField(default=False)

    def __str__(self):
        return self.sarlavha

Django REST Framework — API

Python (DRF) 
# serializers.py
from rest_framework import serializers
from .models import Kurs

class KursSerializer(serializers.ModelSerializer):
    class Meta:
        model = Kurs
        fields = ['id', 'sarlavha', 'tavsif', 'daraja', 'yaratildi']
        read_only_fields = ['yaratildi']

# views.py
from rest_framework import viewsets, permissions
from rest_framework.decorators import action
from rest_framework.response import Response

class KursViewSet(viewsets.ModelViewSet):
    queryset = Kurs.objects.filter(nashr_etildi=True)
    serializer_class = KursSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly]

    @action(detail=False, methods=['get'])
    def mashhur(self, request):
        """Eng mashhur kurslar"""
        kurslar = self.queryset.order_by('-ko_rilgan')[:5]
        serializer = self.get_serializer(kurslar, many=True)
        return Response(serializer.data)

# urls.py
from rest_framework.routers import DefaultRouter
router = DefaultRouter()
router.register('kurslar', KursViewSet)
urlpatterns = router.urls
