Redis Nima?

⏱ 8 daqiqa 🟢 Boshlang'ich 👁 9,400 marta ko'rilgan

Redis (Remote Dictionary Server) — in-memory ma'lumotlar tuzilmasi do'koni. U ma'lumotlarni disk o'rniga operativ xotirada (RAM) saqlaydi, shu sababli juda tez ishlaydi.

Redis kesh (cache), sessiya boshqaruvi, real-vaqtli analitika, navbatlar (queues) va boshqa ko'p hollarda ishlatiladi.

💡 Qachon ishlatish kerak? Tez-tez o'qiladigan, kamdan-kam o'zgaradigan ma'lumotlar uchun ideal — masalan, foydalanuvchi sessiyalari, mahsulot ro'yxatlari, API javoblari.

Redis O'rnatish

Redis Linux, macOS va Windows (WSL) da ishlaydi:

bash 
# Ubuntu/Debian
sudo apt update
sudo apt install redis-server

# Redis ni ishga tushirish
sudo systemctl start redis
sudo systemctl enable redis

# Tekshirish
redis-cli ping
# Javob: PONG

# macOS (Homebrew)
brew install redis
brew services start redis

Redis CLI

Redis CLI orqali Redis serveriga ulaning va buyruqlarni to'g'ridan-to'g'ri kiriting:

bash 
# Redis CLI ni ochish
redis-cli

# Yoki port ko'rsatib
redis-cli -h 127.0.0.1 -p 6379

# Autentifikatsiya (parol bo'lsa)
redis-cli -a yourpassword

# Barcha kalitlarni ko'rish
KEYS *

# Ma'lumot turini bilish
TYPE mykey

# Kalitni o'chirish
DEL mykey

# Serverga ping
PING

Ma'lumot Turlari

Redis quyidagi ma'lumot turlarini qo'llab-quvvatlaydi:

  • 🔤 Strings — oddiy qiymatlar (matn, raqam)
  • 📋 Lists — tartibli elementlar ro'yxati
  • #️⃣ Hashes — kalit-qiymat juftliklari (obyekt kabi)
  • 🔵 Sets — takrorlanmaydigan elementlar to'plami
  • 🏆 Sorted Sets — ball (score) bilan tartiblangan to'plam
  • 📡 Pub/Sub — xabar almashish kanallari

Strings: SET / GET

Strings — Redis da eng oddiy va keng tarqalgan ma'lumot turi:

redis 
# Qiymat saqlash
SET username "Ali"
# OK

# Qiymatni olish
GET username
# "Ali"

# Vaqtinchalik saqlash (10 soniya)
SET token "abc123" EX 10

# Faqat mavjud bo'lmasa o'rnatish
SET counter 1 NX

# Hozirgi qiymatni olish va yangi qiymat o'rnatish
GETSET username "Vali"
# "Ali" (eski qiymat)

# Qiymat uzunligi
STRLEN username
# 4

EXPIRE / TTL

Kalitlarga amal qilish muddati (TTL — Time To Live) belgilash:

redis 
SET session_token "xyz789"

# 3600 soniya (1 soat) muddati belgilash
EXPIRE session_token 3600

# Qolgan vaqtni ko'rish (soniyada)
TTL session_token
# 3598

# Muddatni olib tashlash (abadiy qilish)
PERSIST session_token

# Millisekundda muddati belgilash
PEXPIRE session_token 10000

# Millisekunda qolgan vaqt
PTTL session_token

Lists: LPUSH / RPUSH

Listlar — tartibli elementlar ro'yxati (Deque tuzilmasi):

redis 
# Chapdan qo'shish
LPUSH tasks "vazifa1" "vazifa2" "vazifa3"

# O'ngdan qo'shish
RPUSH tasks "vazifa4"

# Ro'yxatni ko'rish (0 dan oxirigacha)
LRANGE tasks 0 -1
# 1) "vazifa3"
# 2) "vazifa2"
# 3) "vazifa1"
# 4) "vazifa4"

# Uzunlik
LLEN tasks
# 4

# Chapdan o'chirish
LPOP tasks
# "vazifa3"

# O'ngdan o'chirish
RPOP tasks
# "vazifa4"

Hashes: HSET / HGET

Hash — obyektga o'xshash kalit-qiymat juftliklari:

redis 
# Hash yaratish
HSET user:1 name "Ali" age 25 city "Toshkent"

# Bitta maydonni olish
HGET user:1 name
# "Ali"

# Barcha maydonlarni olish
HGETALL user:1
# 1) "name"
# 2) "Ali"
# 3) "age"
# 4) "25"
# 5) "city"
# 6) "Toshkent"

# Maydon mavjudligini tekshirish
HEXISTS user:1 email
# 0 (mavjud emas)

# Maydonni o'chirish
HDEL user:1 city

# Barcha maydon nomlari
HKEYS user:1

Sets: SADD / SMEMBERS

redis 
# Set yaratish (takrorlanmaydi)
SADD hobbies "kitob o'qish" "dasturlash" "sport"

# Barcha elementlar
SMEMBERS hobbies
# 1) "sport"
# 2) "dasturlash"
# 3) "kitob o'qish"

# Element mavjudligini tekshirish
SISMEMBER hobbies "sport"
# 1 (bor)

# Set A va Set B kesishmasi
SADD set1 "a" "b" "c"
SADD set2 "b" "c" "d"
SINTER set1 set2
# "b", "c"

# Birlashma
SUNION set1 set2
# "a", "b", "c", "d"

Sorted Sets: ZADD / ZRANGE

Sorted Sets — har bir elementga ball (score) beriladi va tartiblanadi:

redis 
# Leaderboard yaratish (score: ball)
ZADD leaderboard 1500 "Ali"
ZADD leaderboard 2300 "Vali"
ZADD leaderboard 1800 "Soli"

# Balldan kamiga tartibda
ZRANGE leaderboard 0 -1 WITHSCORES
# 1) "Ali"      2) "1500"
# 3) "Soli"     4) "1800"
# 5) "Vali"     6) "2300"

# Ko'pdan kamiga
ZREVRANGE leaderboard 0 -1 WITHSCORES

# Foydalanuvchi reytingini bilish
ZRANK leaderboard "Ali"
# 0 (birinchi o'rin)

# Ball qo'shish
ZINCRBY leaderboard 500 "Ali"

Pub/Sub

Redis Pub/Sub — real-vaqtli xabar almashish tizimi:

redis 
# Terminal 1 — Obunachi (subscriber)
SUBSCRIBE notifications

# Terminal 2 — Nashriyotchi (publisher)
PUBLISH notifications "Yangi xabar bor!"

# Terminal 1 da natija:
# 1) "message"
# 2) "notifications"
# 3) "Yangi xabar bor!"

# Pattern bilan obuna bo'lish
PSUBSCRIBE user:*

# Barcha kanallarni ko'rish
PUBSUB CHANNELS

# Obunani bekor qilish
UNSUBSCRIBE notifications

Caching Pattern

Redis kesh sifatida ishlatish — eng keng tarqalgan foydalanish holati:

javascript 
// Node.js + ioredis bilan kesh pattern
const Redis = require('ioredis');
const redis = new Redis();

async function getUserData(userId) {
  const cacheKey = `user:${userId}`;
  
  // 1. Avval keshdan tekshir
  const cached = await redis.get(cacheKey);
  if (cached) {
    console.log('Keshdan olindi ✅');
    return JSON.parse(cached);
  }
  
  // 2. Ma'lumotlar bazasidan ol
  const user = await db.query('SELECT * FROM users WHERE id = ?', [userId]);
  
  // 3. Keshga saqlash (1 soat)
  await redis.set(cacheKey, JSON.stringify(user), 'EX', 3600);
  
  console.log('DB dan olindi 📦');
  return user;
}

// Cache invalidation (yangilanishda o'chirish)
async function updateUser(userId, data) {
  await db.update('users', data, userId);
  await redis.del(`user:${userId}`); // Keshni o'chirish
}

Node.js bilan Redis

javascript 
// npm install ioredis
const Redis = require('ioredis');

const redis = new Redis({
  host: '127.0.0.1',
  port: 6379,
  // password: 'your_password',
});

// Asosiy amallar
async function main() {
  // String
  await redis.set('name', 'Ali', 'EX', 60);
  const name = await redis.get('name');
  console.log(name); // Ali

  // Hash
  await redis.hset('user:1', 'name', 'Ali', 'age', '25');
  const user = await redis.hgetall('user:1');
  console.log(user); // { name: 'Ali', age: '25' }

  // List
  await redis.rpush('queue', 'task1', 'task2');
  const task = await redis.lpop('queue');
  console.log(task); // task1

  // Ulanishni yopish
  redis.disconnect();
}

main().catch(console.error);
