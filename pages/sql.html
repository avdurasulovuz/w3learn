SQL Nima?

⏱ 10 daqiqa 🟢 Boshlang'ich 👁 29,500 marta ko'rilgan

SQL (Structured Query Language) — relyatsion ma'lumotlar bazalarini boshqarish uchun standart til. 1974-yilda IBM tomonidan yaratilgan.

SQL yordamida ma'lumotlar yaratish, o'qish, yangilash va o'chirish (CRUD) amallarini bajarish mumkin. MySQL, PostgreSQL, SQLite, SQL Server, Oracle — bularning barchasi SQL dan foydalanadi.

💡 SQL turlari: DDL (tuzilma), DML (ma'lumot), DCL (ruxsat), TCL (tranzaksiya).

Jadval Yaratish (CREATE TABLE)

sql 
-- Ma'lumotlar bazasi yaratish
CREATE DATABASE maktab;
USE maktab;

-- Jadval yaratish
CREATE TABLE talabalar (
    id          INT PRIMARY KEY AUTO_INCREMENT,
    ism         VARCHAR(50) NOT NULL,
    familiya    VARCHAR(50) NOT NULL,
    email       VARCHAR(100) UNIQUE,
    yoshi       INT CHECK (yoshi >= 16 AND yoshi <= 60),
    kurs        TINYINT DEFAULT 1,
    tug_sana    DATE,
    yaratilgan  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Bog'liq jadval (Foreign Key)
CREATE TABLE kurslar (
    id    INT PRIMARY KEY AUTO_INCREMENT,
    nomi  VARCHAR(100) NOT NULL,
    narxi DECIMAL(10, 2)
);

CREATE TABLE yozilishlar (
    id         INT PRIMARY KEY AUTO_INCREMENT,
    talaba_id  INT NOT NULL,
    kurs_id    INT NOT NULL,
    sana       DATE DEFAULT (CURRENT_DATE),
    FOREIGN KEY (talaba_id) REFERENCES talabalar(id),
    FOREIGN KEY (kurs_id) REFERENCES kurslar(id)
);

Ma'lumot Qo'shish (INSERT)

sql 
-- Bitta qator qo'shish
INSERT INTO talabalar (ism, familiya, email, yoshi, kurs)
VALUES ('Ali', 'Karimov', 'ali@mail.uz', 20, 2);

-- Ko'p qator qo'shish
INSERT INTO talabalar (ism, familiya, yoshi, kurs)
VALUES 
  ('Vali', 'Rahimov', 22, 3),
  ('Soli', 'Hasanov', 19, 1),
  ('Nozima', 'Yusupova', 21, 2);

-- SELECT dan INSERT
INSERT INTO arxiv_talabalar
SELECT * FROM talabalar WHERE kurs = 4;

-- UPSERT (mavjud bo'lsa yangilash)
INSERT INTO talabalar (email, ism, familiya)
VALUES ('ali@mail.uz', 'Ali', 'Karimov')
ON DUPLICATE KEY UPDATE ism = VALUES(ism);

Ma'lumot O'qish (SELECT)

sql 
-- Barcha ustunlar
SELECT * FROM talabalar;

-- Tanlangan ustunlar
SELECT ism, familiya, email FROM talabalar;

-- Taxallus (alias)
SELECT 
    ism AS "Ismi",
    familiya AS "Familiyasi",
    yoshi AS "Yoshi"
FROM talabalar;

-- Hisoblash
SELECT 
    ism,
    yoshi,
    yoshi + 5 AS "5 yildan keyin"
FROM talabalar;

-- DISTINCT — takrorlanmaslar
SELECT DISTINCT kurs FROM talabalar;

-- Shart bilan
SELECT * FROM talabalar WHERE kurs = 2;
SELECT * FROM talabalar WHERE yoshi BETWEEN 18 AND 25;
SELECT * FROM talabalar WHERE ism LIKE 'A%';
SELECT * FROM talabalar WHERE kurs IN (1, 2, 3);
SELECT * FROM talabalar WHERE email IS NOT NULL;

Yangilash va O'chirish

sql 
-- UPDATE — yangilash
UPDATE talabalar
SET kurs = 3, email = 'yangi@mail.uz'
WHERE id = 1;

-- Ko'p qatorni yangilash
UPDATE talabalar
SET kurs = kurs + 1
WHERE kurs < 4;

-- ⚠️ Ehtiyot: WHERE siz BARCHA qatorlar yangilanadi!

-- DELETE — o'chirish
DELETE FROM talabalar WHERE id = 5;

-- Shartli o'chirish
DELETE FROM talabalar 
WHERE kurs = 4 AND yoshi > 30;

-- TRUNCATE — jadvalning barcha ma'lumotlarini o'chirish (tez)
TRUNCATE TABLE log_jadval;

-- ⚠️ DROP — jadvalning o'zi bilan ma'lumotlar
DROP TABLE eski_jadval;

Agregat Funksiyalar

sql 
-- Soni
SELECT COUNT(*) AS "Talabalar soni" FROM talabalar;
SELECT COUNT(email) AS "Emailli talabalar" FROM talabalar;

-- Yig'indisi
SELECT SUM(narxi) AS "Jami narx" FROM kurslar;

-- O'rtacha
SELECT AVG(yoshi) AS "O'rtacha yosh" FROM talabalar;

-- Maksimum / Minimum
SELECT MAX(yoshi), MIN(yoshi) FROM talabalar;

-- GROUP BY — guruhlab hisoblash
SELECT kurs, COUNT(*) AS soni, AVG(yoshi) AS ort_yosh
FROM talabalar
GROUP BY kurs
ORDER BY kurs;

-- HAVING — guruh uchun shart (WHERE emas!)
SELECT kurs, COUNT(*) AS soni
FROM talabalar
GROUP BY kurs
HAVING COUNT(*) > 5
ORDER BY soni DESC;

JOIN — Jadvallarni Birlashtirish

sql 
-- INNER JOIN — ikkala jadvalda mavjud bo'lganlar
SELECT t.ism, t.familiya, k.nomi AS kurs_nomi
FROM talabalar t
INNER JOIN yozilishlar y ON t.id = y.talaba_id
INNER JOIN kurslar k ON y.kurs_id = k.id;

-- LEFT JOIN — chap jadvalning barchasi + o'ng moslar
SELECT t.ism, k.nomi
FROM talabalar t
LEFT JOIN yozilishlar y ON t.id = y.talaba_id
LEFT JOIN kurslar k ON y.kurs_id = k.id;
-- Hech qaysi kursga yozilmagan talabalar ham chiqadi (NULL bilan)

-- Murakkab JOIN misoli
SELECT 
    t.ism || ' ' || t.familiya AS to_liq_ism,
    COUNT(y.kurs_id) AS yozilgan_kurslar,
    GROUP_CONCAT(k.nomi SEPARATOR ', ') AS kurs_nomlari
FROM talabalar t
LEFT JOIN yozilishlar y ON t.id = y.talaba_id
LEFT JOIN kurslar k ON y.kurs_id = k.id
GROUP BY t.id
ORDER BY yozilgan_kurslar DESC;

Subquery va VIEW

sql 
-- Subquery — ichki so'rov
SELECT ism, yoshi
FROM talabalar
WHERE yoshi > (SELECT AVG(yoshi) FROM talabalar);

-- IN bilan subquery
SELECT ism FROM talabalar
WHERE id IN (
    SELECT talaba_id FROM yozilishlar
    WHERE kurs_id = 3
);

-- EXISTS
SELECT ism FROM talabalar t
WHERE EXISTS (
    SELECT 1 FROM yozilishlar y WHERE y.talaba_id = t.id
);

-- VIEW — virtual jadval
CREATE VIEW faol_talabalar AS
SELECT t.ism, t.familiya, COUNT(y.id) AS kurslar_soni
FROM talabalar t
JOIN yozilishlar y ON t.id = y.talaba_id
GROUP BY t.id
HAVING kurslar_soni > 0;

-- VIEW dan foydalanish
SELECT * FROM faol_talabalar ORDER BY kurslar_soni DESC;

-- INDEX — tezlashtirishuchun
CREATE INDEX idx_email ON talabalar(email);
CREATE INDEX idx_kurs_yosh ON talabalar(kurs, yoshi);

Tranzaksiyalar

sql 
-- Tranzaksiya — atomik amallar guruhi
BEGIN TRANSACTION;

-- Bank o'tkazmasi misoli
UPDATE hisoblar SET balans = balans - 500 WHERE id = 1;
UPDATE hisoblar SET balans = balans + 500 WHERE id = 2;

-- Agar muvaffaqiyatli
COMMIT;

-- Agar xato yuz berdi — bekor qilish
ROLLBACK;

-- SAVEPOINT — qisman bekor qilish
BEGIN;
INSERT INTO talabalar (ism) VALUES ('Test1');
SAVEPOINT sp1;
INSERT INTO talabalar (ism) VALUES ('Test2');
ROLLBACK TO SAVEPOINT sp1;  -- faqat Test2 bekor
COMMIT;  -- Test1 saqlanadi
