C++ Nima?

⏱ 12 daqiqa 🔴 Ilg'or 👁 4,230 marta ko'rilgan

C++ — Bjarne Stroustrup tomonidan yaratilgan, C tilining kengaytmasi. Tizim dasturlash, o'yinlar, brauzerlar va yuqori samarali dasturlar yaratishda eng ko'p ishlatiladigan tillardan biri.

💡 Eslatma: C++ — past darajali (low-level) va yuqori darajali (high-level) xususiyatlarni birlashtiradi.

Asosiy Sintaksis

C++ 
#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
    // Chiqarish
    cout << "Salom, Dunyo!" << endl;

    // O'zgaruvchilar
    string ism = "Ali";
    int yosh = 25;
    double baland = 1.75;
    bool faol = true;

    cout << "Ism: " << ism << ", Yosh: " << yosh << endl;

    // Vector (dinamik massiv)
    vector<int> sonlar = {1, 2, 3, 4, 5};
    for (int son : sonlar) {
        cout << son << " ";
    }
    cout << endl;

    // Funksiya chaqirish
    return 0;
}

Pointerlar

C++ (Pointerlar) 
#include <iostream>
using namespace std;

int main() {
    int son = 42;

    // Pointer — xotira manziliga ko'rsatuvchi
    int* ptr = &son;  // & — manzilini olish

    cout << "Qiymat: " << son << endl;
    cout << "Manzil: " << &son << endl;
    cout << "Pointer: " << ptr << endl;
    cout << "Dereference: " << *ptr << endl; // * — qiymatni olish

    // Qiymatni pointer orqali o'zgartirish
    *ptr = 100;
    cout << "Yangi qiymat: " << son << endl; // 100

    // Dinamik xotira
    int* dinamik = new int(55);
    cout << "Dinamik: " << *dinamik << endl;
    delete dinamik; // xotirani ozod qilish

    return 0;
}

OOP — Klasslar

C++ (OOP) 
#include <iostream>
#include <string>
using namespace std;

class Foydalanuvchi {
private:
    string ism;
    string email;
protected:
    int yosh;

public:
    // Konstruktor
    Foydalanuvchi(string ism, string email, int yosh)
        : ism(ism), email(email), yosh(yosh) {}

    // Destruktor
    ~Foydalanuvchi() {}

    // Metodlar
    string getIsm() const { return ism; }
    string getMalumat() const {
        return ism + " (" + email + "), yosh: " + to_string(yosh);
    }

    bool voyagaYetganmi() const { return yosh >= 18; }
};

// Meros
class Admin : public Foydalanuvchi {
private:
    string daraja;

public:
    Admin(string ism, string email, int yosh, string daraja)
        : Foydalanuvchi(ism, email, yosh), daraja(daraja) {}

    string getDaraja() const { return daraja; }
};

int main() {
    Foydalanuvchi user("Ali", "ali@test.com", 25);
    cout << user.getMalumat() << endl;

    Admin admin("Sarvar", "sarvar@test.com", 30, "Super Admin");
    cout << admin.getMalumat() << " — " << admin.getDaraja() << endl;

    return 0;
}
