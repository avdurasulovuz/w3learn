Node.js Nima?

⏱ 10 daqiqa 🟡 O'rta 👁 7,560 marta ko'rilgan

Node.js — JavaScript ni server tomonida ishlatish imkonini beruvchi muhit (runtime). Google Chrome ning V8 dvigatelida ishlaydi. Node.js yordamida backend API, real-time ilovalar va mikroservislar yaratish mumkin.

Node.js asinxron va event-driven arxitekturada ishlaydi, bu esa uni yuqori yuklamalarni samarali boshqarishi imkonini beradi.

💡 Eslatma: Node.js o'rnatilgandan so'ng node --version buyrug'i bilan versiyani tekshiring.

Oddiy HTTP Server

JavaScript (Node.js) 
// server.js
const http = require('http');

const server = http.createServer((req, res) => {
  // HTTP javob sarlavhalari
  res.writeHead(200, { 'Content-Type': 'text/plain; charset=utf-8' });

  // Marshrutlash
  if (req.url === '/') {
    res.end('Bosh sahifa — Salom, Node.js!');
  } else if (req.url === '/haqida') {
    res.end('Biz haqimizda sahifasi');
  } else {
    res.writeHead(404);
    res.end('404 — Sahifa topilmadi');
  }
});

// 3000-portda tinglash
server.listen(3000, () => {
  console.log('Server ishga tushdi: http://localhost:3000');
});

Express.js bilan REST API

JavaScript (Express) 
// npm install express
const express = require('express');
const app = express();

// JSON ma'lumotlarni o'qish
app.use(express.json());

// Foydalanuvchilar ma'lumotlari (misol)
let foydalanuvchilar = [
  { id: 1, ism: 'Ali', email: 'ali@example.com' },
  { id: 2, ism: 'Zulfiya', email: 'zulfiya@example.com' },
];

// GET — barcha foydalanuvchilar
app.get('/api/users', (req, res) => {
  res.json({ success: true, data: foydalanuvchilar });
});

// GET — bitta foydalanuvchi
app.get('/api/users/:id', (req, res) => {
  const user = foydalanuvchilar.find(u => u.id == req.params.id);
  if (!user) return res.status(404).json({ error: 'Topilmadi' });
  res.json({ success: true, data: user });
});

// POST — yangi foydalanuvchi qo'shish
app.post('/api/users', (req, res) => {
  const yangi = { id: Date.now(), ...req.body };
  foydalanuvchilar.push(yangi);
  res.status(201).json({ success: true, data: yangi });
});

app.listen(3000, () => console.log('API: http://localhost:3000'));

JWT Token bilan Autentifikatsiya

JavaScript (JWT) 
// npm install jsonwebtoken bcryptjs
const jwt = require('jsonwebtoken');
const bcrypt = require('bcryptjs');

const SECRET = process.env.JWT_SECRET || 'maxfiy_kalit';

// Parolni shifrlash
async function parolShifrla(parol) {
  return await bcrypt.hash(parol, 10);
}

// Login endpoint
app.post('/api/login', async (req, res) => {
  const { email, parol } = req.body;

  // Foydalanuvchini topish (MB dan)
  const user = await User.findOne({ email });
  if (!user) return res.status(401).json({ error: 'Noto\'g\'ri email' });

  // Parolni tekshirish
  const togri = await bcrypt.compare(parol, user.parolHash);
  if (!togri) return res.status(401).json({ error: 'Noto\'g\'ri parol' });

  // JWT token yaratish
  const token = jwt.sign(
    { id: user._id, email: user.email },
    SECRET,
    { expiresIn: '7d' }
  );

  res.json({ success: true, token });
});
