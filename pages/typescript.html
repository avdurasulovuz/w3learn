TypeScript Nima?

⏱ 10 daqiqa 🔴 Ilg'or 👁 4,780 marta ko'rilgan

TypeScript — Microsoft tomonidan yaratilgan JavaScript ning kengaytmasi (superset). TypeScript statik tiplash ni qo'shib, katta loyihalarda xatolarni oldindan aniqlash imkonini beradi.

💡 Eslatma: Har qanday JavaScript kodi — to'g'ri TypeScript koddir. TS ni tsc kompilyatori JS ga aylantiradi.

Asosiy Typlar

TypeScript 
// Primitiv typlar
let ism: string = "Ali";
let yosh: number = 25;
let faol: boolean = true;

// Array
let sonlar: number[] = [1, 2, 3, 4, 5];
let ismlar: Array<string> = ["Ali", "Zulfiya", "Kamol"];

// Tuple (aniq uzunlik va typlar)
let koordinata: [number, number] = [41.2995, 69.2401];

// Union type
let id: number | string = 123;
id = "ABC-456"; // ham ishlaydi

// Optional
function salomlash(ism: string, unvon?: string): string {
  return unvon ? `${unvon} ${ism}` : ism;
}

console.log(salomlash("Ali"));           // Ali
console.log(salomlash("Ali", "Janob")); // Janob Ali

Interface

TypeScript 
// Interface — obyekt tuzilmasini belgilash
interface Foydalanuvchi {
  id: number;
  ism: string;
  email: string;
  yosh?: number;         // optional
  readonly yaratildi: Date; // faqat o'qish
}

// Interface ni kengaytirish
interface Admin extends Foydalanuvchi {
  ruxsatlar: string[];
  darajasi: "super" | "oddiy";
}

// Funksiya interface
interface ApiJavob<T> {
  success: boolean;
  data: T;
  xabar?: string;
}

// Ishlatish
const user: Foydalanuvchi = {
  id: 1,
  ism: "Ali Karimov",
  email: "ali@test.com",
  yaratildi: new Date()
};

const javob: ApiJavob<Foydalanuvchi> = {
  success: true,
  data: user
};

Generics

TypeScript 
// Generic funksiya — har qanday type bilan ishlaydi
function birinchi<T>(arr: T[]): T {
  return arr[0];
}

console.log(birinchi([1, 2, 3]));       // 1 (number)
console.log(birinchi(["a", "b", "c"])); // "a" (string)

// Generic class
class Stack<T> {
  private items: T[] = [];

  push(item: T): void {
    this.items.push(item);
  }

  pop(): T | undefined {
    return this.items.pop();
  }

  get uzunlik(): number {
    return this.items.length;
  }
}

const sonStack = new Stack<number>();
sonStack.push(10);
sonStack.push(20);
console.log(sonStack.pop()); // 20
