PostgreSQL Nima?

⏱ 8 daqiqa 🟡 O'rta 👁 4,560 marta ko'rilgan

PostgreSQL — eng kuchli ochiq manbali relyatsion ma'lumotlar bazasi (RDBMS). MySQL dan farqli ravishda JSONB, massivlar, to'liq matn qidiruvi va ilg'or funksiyalarni qo'llaydi.

💡 Eslatma: PostgreSQL — ACID tranzaksiyalarni to'liq qo'llaydi va yuqori darajada kengaytiriladigan (extensible) MB.

Jadval Yaratish

SQL (PostgreSQL) 
-- Ma'lumotlar bazasi yaratish
CREATE DATABASE w3learn_db;

-- Jadval yaratish
CREATE TABLE foydalanuvchilar (
    id          SERIAL PRIMARY KEY,
    ism         VARCHAR(100) NOT NULL,
    email       VARCHAR(255) UNIQUE NOT NULL,
    yosh        INTEGER CHECK (yosh >= 0 AND yosh <= 150),
    shahar      VARCHAR(100) DEFAULT 'Noma''lum',
    konikmalari TEXT[],              -- Massiv type
    malumot     JSONB,               -- JSON type
    yaratildi   TIMESTAMPTZ DEFAULT NOW(),
    yangilandi  TIMESTAMPTZ DEFAULT NOW()
);

-- Index yaratish (tezlashtirish uchun)
CREATE INDEX idx_email ON foydalanuvchilar(email);
CREATE INDEX idx_ism ON foydalanuvchilar(LOWER(ism));

-- Ma'lumot qo'shish
INSERT INTO foydalanuvchilar (ism, email, yosh, shahar, konikmalari, malumot)
VALUES

     ARRAY['JavaScript', 'React'], '{"daraja": "senior"}'::jsonb),
    ('Zulfiya Rahimova', 'zulfiya@test.com', 22, 'Samarqand',
     ARRAY['Python', 'Django'], '{"daraja": "junior"}'::jsonb);

JOIN turlari

SQL (PostgreSQL) 
-- Jadvallar
CREATE TABLE kurslar (
    id      SERIAL PRIMARY KEY,
    nomi    VARCHAR(200) NOT NULL,
    narxi   DECIMAL(10,2)
);

CREATE TABLE buyurtmalar (
    id              SERIAL PRIMARY KEY,
    foydalanuvchi_id INTEGER REFERENCES foydalanuvchilar(id),
    kurs_id         INTEGER REFERENCES kurslar(id),
    sana            DATE DEFAULT CURRENT_DATE
);

-- INNER JOIN
SELECT f.ism, k.nomi AS kurs, b.sana
FROM buyurtmalar b
INNER JOIN foydalanuvchilar f ON b.foydalanuvchi_id = f.id
INNER JOIN kurslar k ON b.kurs_id = k.id
ORDER BY b.sana DESC;

-- LEFT JOIN (hamma foydalanuvchilar, hatto buyurtma yo'q bo'lsa ham)
SELECT f.ism, COUNT(b.id) AS buyurtmalar_soni
FROM foydalanuvchilar f
LEFT JOIN buyurtmalar b ON f.id = b.foydalanuvchi_id
GROUP BY f.id, f.ism
ORDER BY buyurtmalar_soni DESC;

JSONB va Window Funksiyalar

SQL (PostgreSQL) 
-- JSONB so'rovlar
SELECT ism, malumot->>'daraja' AS daraja
FROM foydalanuvchilar
WHERE malumot->>'daraja' = 'senior';

-- Massiv bilan ishlash
SELECT ism, konikmalari
FROM foydalanuvchilar
WHERE 'JavaScript' = ANY(konikmalari);

-- Window funksiyalar
SELECT
    ism, shahar, yosh,
    AVG(yosh) OVER (PARTITION BY shahar) AS shahar_ortacha_yosh,
    RANK() OVER (PARTITION BY shahar ORDER BY yosh DESC) AS shahar_reyting,
    ROW_NUMBER() OVER (ORDER BY yaratildi) AS tartib_raqam
FROM foydalanuvchilar;

-- CTE (Common Table Expression)
WITH yoshlar AS (
    SELECT shahar, AVG(yosh) AS ortacha
    FROM foydalanuvchilar
    GROUP BY shahar
)
SELECT f.ism, f.shahar, f.yosh, y.ortacha
FROM foydalanuvchilar f
JOIN yoshlar y ON f.shahar = y.shahar
WHERE f.yosh > y.ortacha;
