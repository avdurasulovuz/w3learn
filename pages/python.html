Python Nima?

⏱ 10 daqiqa 🟢 Boshlang'ich 👁 38,700 marta ko'rilgan

Python — oddiy sintaksisga ega, kuchli va ko'p qirrali dasturlash tili. Guido van Rossum tomonidan 1991-yilda yaratilgan.

Veb dasturlash (Django, Flask), ma'lumotlar ilmi (pandas, numpy), sun'iy intellekt (TensorFlow), avtomatlashtirish va boshqa ko'p sohalarda qo'llaniladi.

💡 Python falsafasi (Zen of Python): "Oddiy yaxshi. Murakkab yomondan yaxshi. O'qilishi muhim."

Tsikllar

python 
# for tsikli
for i in range(5):
    print(i)  # 0, 1, 2, 3, 4

# range(start, stop, step)
for i in range(1, 10, 2):
    print(i)  # 1, 3, 5, 7, 9

# Ro'yxat bo'ylab
tillar = ["Python", "JS", "Java"]
for til in tillar:
    print(f"🔹 {til}")

# enumerate — indeks bilan
for i, til in enumerate(tillar):
    print(f"{i+1}. {til}")

# while tsikli
son = 1
while son <= 5:
    print(son)
    son += 1

# List Comprehension
kvadratlar = [x**2 for x in range(1, 6)]
print(kvadratlar)  # [1, 4, 9, 16, 25]

juftlar = [x for x in range(20) if x % 2 == 0]
print(juftlar)  # [0, 2, 4, 6, ..., 18]

Funksiyalar

python 
# Oddiy funksiya
def salomlash(ism):
    return f"Salom, {ism}!"

print(salomlash("Ali"))  # Salom, Ali!

# Default parametr
def quvvat(son, daraja=2):
    return son ** daraja

print(quvvat(3))     # 9
print(quvvat(2, 10)) # 1024

# *args — ixtiyoriy miqdordagi argumentlar
def yigindi(*sonlar):
    return sum(sonlar)

print(yigindi(1, 2, 3, 4, 5))  # 15

# **kwargs — kalit-qiymat argumentlar
def profil(**ma_lumot):
    for kalit, qiymat in ma_lumot.items():
        print(f"{kalit}: {qiymat}")

profil(ism="Ali", yosh=25, shahar="Toshkent")

# Lambda (anonim funksiya)
kvadrat = lambda x: x ** 2
print(kvadrat(5))  # 25

# sorted bilan lambda
odamlar = [{"ism": "Ali", "yosh": 25}, {"ism": "Vali", "yosh": 20}]
sortlangan = sorted(odamlar, key=lambda x: x["yosh"])
print(sortlangan)

python 
foydalanuvchi = {
    "ism": "Ali",
    "yosh": 25,
    "email": "ali@example.com",
    "faol": True
}

foydalanuvchi.update({"yosh": 26, "lavozim": "Dasturchi"})

# O'chirish
del foydalanuvchi["faol"]
yosh = foydalanuvchi.pop("yosh")

    print(f"{kalit}: {foydalanuvchi[kalit]}")

for kalit, qiymat in foydalanuvchi.items():
    print(f"{kalit} = {qiymat}")

# Dict Comprehension
kvadratlar = {x: x**2 for x in range(1, 6)}
print(kvadratlar)  # {1:1, 2:4, 3:9, 4:16, 5:25}

OOP — Sinflar

python 
class Avtomobil:
    # Sinf atributi
    g'ildirak_soni = 4
    
    def __init__(self, marka, model, yil):
        self.marka = marka
        self.model = model
        self.yil = yil
        self.__tezlik = 0  # private
    
    def tezlashtir(self, miqdor):
        self.__tezlik += miqdor
        return self
    
    def __str__(self):
        return f"{self.yil} {self.marka} {self.model}"
    
    @property
    def tezlik(self):
        return self.__tezlik


# Meros (Inheritance)
class ElektrikAvto(Avtomobil):
    def __init__(self, marka, model, yil, batareya_quvvati):
        super().__init__(marka, model, yil)
        self.batareya = batareya_quvvati
    
    def quvvatlash(self):
        return f"{self.marka} quvvatlanmoqda... ⚡"


tesla = ElektrikAvto("Tesla", "Model 3", 2023, 75)
print(tesla)              # 2023 Tesla Model 3
print(tesla.quvvatlash()) # Tesla quvvatlanmoqda... ⚡
tesla.tezlashtir(60)
print(tesla.tezlik)       # 60
